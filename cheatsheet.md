FROM     — откуда (таблица)
SELECT   — какие столбцы (или * = все)
WHERE    — какие строки (условие)
LIKE     — поиск по шаблону
NOT LIKE — не соответствует шаблону
%        — любое количество символов (включая ноль)
_        — ровно один любой символ
IN       — строгое соответствие списку ('A', 'B', 'C')
NOT IN   — строгое несоответствие списку
BETWEEN  — между двумя значениями (включительно)
AND      — «и» (оба условия)
OR       — «или» (достаточно одного)
'...'    — строки (одинарные кавычки)
числа    — без кавычек
